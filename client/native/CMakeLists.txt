cmake_minimum_required(VERSION 3.20)
project(QuickShareProcessAudio LANGUAGES CXX)
add_executable(process-audio process-audio.cpp)
target_compile_features(process-audio PRIVATE cxx_std_17)
target_compile_definitions(process-audio PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(process-audio PRIVATE ole32 mmdevapi user32)
if(MSVC)
  target_compile_options(process-audio PRIVATE /W4 /EHsc /utf-8)
  set_property(TARGET process-audio PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set_target_properties(process-audio PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/bin")
